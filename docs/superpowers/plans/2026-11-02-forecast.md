# Forecast Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A forecast engine driven by forecast-only templates, invoices, budgets/history and credit-card cycles, exposed through an API, a `query_forecast` MCP tool and one overview widget (month-end cash with/without expected income, shortfalls). The full Forecast page, the 3-month/P&L views in the UI and suggest-from-history are deferred (trimmed 2026-11-02; the API already returns 1–3 months and P&L fields).

**Architecture:** All forecasting maths lives in a new pure Go package `pkg/forecast` (plain structs in, plain structs out, no DB). `pkg/services/forecast.go` loads data from existing services, maps it into `forecast.Input`, and calls `forecast.Build`. One API endpoint and one MCP tool expose the result; an overview widget shows this month.

**Tech Stack:** Go (xorm, gin), Vue 3 + Vuetify (desktop) / Framework7 (mobile), vitest, `go test`.

**Spec:** `docs/superpowers/specs/2026-11-02-forecast-design.md`

## Global Constraints

- Schema changes are new columns/enum values only; existing rows keep today's behaviour.
- Amounts are int64 minor units (paise). Dates inside `pkg/forecast` are local calendar dates as `time.Time` at 00:00 UTC (`forecast.D(2026, 11, 5)`).
- New template type `TRANSACTION_TEMPLATE_TYPE_FORECAST = 3`; it must never post a transaction.
- Matching tolerance ±20%; one-off = ≥ 3× category median; card bill fulfilled by a repayment ≥ 90% of the billed amount.
- `months` query parameter 1–3.
- Accounts negative at the start are never shortfall-alerted.
- Desktop + mobile both; shared logic in `src/lib/forecast.ts`.
- Commits: `git commit --no-verify` (the beads pre-commit hook is broken). Vitest: `npx vitest run --exclude '.claude/**'`.

```go
// pkg/forecast/types.go
// Package forecast turns planned items, invoices, budgets, history and card cycles into a dated cash forecast.
// It is pure: callers map database rows into these structs.
package forecast

import "time"

type Kind byte

const (
	KindExpense  Kind = 1
	KindIncome   Kind = 2
	KindTransfer Kind = 3
)

type Source string

const (
	SourceTemplate         Source = "template"
	SourceInvoice          Source = "invoice"
	SourceRecurringInvoice Source = "recurring_invoice"
	SourceEveryday         Source = "everyday"
	SourceCardBill         Source = "card_bill"
)

// Frequency types mirror models.TransactionScheduleFrequencyType values
const (
	FreqWeekly    byte = 1
	FreqMonthly   byte = 2
	FreqDaily     byte = 3
	FreqYearly    byte = 4
	FreqEveryNDay byte = 5
)

// Template is a forecast-only transaction template
type Template struct {
	Id, AccountId, DestinationAccountId, CategoryId int64
	Kind                                            Kind
	Amount                                          int64
	Business                                        bool
	Label                                           string
	FrequencyType                                   byte
	Frequency                                       string    // as stored: "5" or "1,15" or "-1"
	Start, End                                      time.Time // zero = unbounded
}

// Txn is a real transaction (transfers: AccountId → DestinationAccountId)
type Txn struct {
	Id                                              int64
	Date                                            time.Time
	Kind                                            Kind
	AccountId, DestinationAccountId, CategoryId     int64
	Amount                                          int64
	Business                                        bool
	PayeeKey                                        string
	CardRepayment                                   bool // transfer category is Credit Card Repayment
}

// Item is one expected money movement
type Item struct {
	Date                                        time.Time
	Kind                                        Kind
	AccountId, DestinationAccountId, CategoryId int64
	Amount                                      int64
	Source                                      Source
	SourceId                                    int64
	Uncertain, Business                         bool
	Label                                       string
}

func D(y int, m time.Month, d int) time.Time { return time.Date(y, m, d, 0, 0, 0, 0, time.UTC) }

func MonthStart(t time.Time) time.Time { return D(t.Year(), t.Month(), 1) }

func NextMonth(t time.Time) time.Time { return MonthStart(t).AddDate(0, 1, 0) }

func daysIn(t time.Time) int { return NextMonth(t).AddDate(0, 0, -1).Day() }
```

```go
// pkg/forecast/schedule.go
package forecast

import (
	"strconv"
	"strings"
	"time"
)

// ExpandTemplate returns one item per scheduled date in [from, to), same rules as the scheduled-transaction cron
func ExpandTemplate(t Template, from, to time.Time) []Item {
	var values []int
	for _, s := range strings.Split(t.Frequency, ",") {
		v, err := strconv.Atoi(strings.TrimSpace(s))
		if err != nil {
			return nil
		}
		values = append(values, v)
	}
	if len(values) == 0 || t.FrequencyType < FreqWeekly || t.FrequencyType > FreqEveryNDay {
		return nil
	}

	var items []Item
	for d := from; d.Before(to); d = d.AddDate(0, 0, 1) {
		if (!t.Start.IsZero() && d.Before(t.Start)) || (!t.End.IsZero() && d.After(t.End)) || !matches(t, values, d) {
			continue
		}
		items = append(items, Item{Date: d, Kind: t.Kind, AccountId: t.AccountId, DestinationAccountId: t.DestinationAccountId,
			CategoryId: t.CategoryId, Amount: t.Amount, Source: SourceTemplate, SourceId: t.Id, Business: t.Business, Label: t.Label})
	}
	return items
}

func matches(t Template, values []int, d time.Time) bool {
	for _, v := range values {
		switch t.FrequencyType {
		case FreqDaily:
			return true
		case FreqWeekly:
			if int(d.Weekday()) == v {
				return true
			}
		case FreqMonthly:
			last := daysIn(d)
			day := v
			if day < 0 {
				day = last + v + 1
			} else if day > last {
				day = last // ponytail: day 29–31 falls on the last day; the cron skips such months instead
			}
			if d.Day() == day {
				return true
			}
		case FreqYearly:
			if int(d.Month())*100+d.Day() == v {
				return true
			}
		case FreqEveryNDay:
			if t.Start.IsZero() || v <= 0 {
				return false
			}
			diff := int(d.Sub(t.Start).Hours() / 24)
			return diff >= 0 && diff%v == 0
		}
	}
	return false
}
```

```go
// pkg/forecast/match.go
package forecast

import "sort"

const matchTolerance = 0.20

// DropFulfilled removes template items that a real transaction in the same month already covers.
// Each transaction fulfils at most one item; the closest date wins.
func DropFulfilled(items []Item, actual []Txn) []Item {
	used := map[int]bool{}
	var kept []Item
	for _, it := range items {
		if it.Source != SourceTemplate {
			kept = append(kept, it)
			continue
		}
		best, bestGap := -1, 0.0
		for i, tx := range actual {
			if used[i] || tx.Kind != it.Kind || tx.AccountId != it.AccountId || tx.CategoryId != it.CategoryId ||
				!MonthStart(tx.Date).Equal(MonthStart(it.Date)) || !within(tx.Amount, it.Amount) {
				continue
			}
			gap := tx.Date.Sub(it.Date).Abs().Hours()
			if best < 0 || gap < bestGap {
				best, bestGap = i, gap
			}
		}
		if best >= 0 {
			used[best] = true
			continue
		}
		kept = append(kept, it)
	}
	sort.SliceStable(kept, func(i, j int) bool { return kept[i].Date.Before(kept[j].Date) })
	return kept
}

func within(actual, expected int64) bool {
	if expected == 0 {
		return actual == 0
	}
	diff := float64(actual-expected) / float64(expected)
	return diff >= -matchTolerance && diff <= matchTolerance
}
```

```go
// pkg/forecast/everyday.go
package forecast

import (
	"sort"
	"time"
)

// Budget is a monthly budget scoped to exactly one expense category
type Budget struct {
	CategoryId int64
	Amount     int64
}

const oneOffFactor = 3

// Everyday estimates non-template spending for month, as one item per (account, business) pair
func Everyday(history []Txn, templates []Template, budgets []Budget, today, month time.Time) []Item {
	histStart := MonthStart(today).AddDate(0, -3, 0)
	histEnd := MonthStart(today)

	type key struct {
		acct     int64
		business bool
	}
	perCat := map[int64][]Txn{}
	spentThisMonth := map[int64]int64{}
	for _, tx := range history {
		if tx.Kind != KindExpense || coveredByTemplate(tx, templates) {
			continue
		}
		if !tx.Date.Before(histStart) && tx.Date.Before(histEnd) {
			perCat[tx.CategoryId] = append(perCat[tx.CategoryId], tx)
		} else if MonthStart(tx.Date).Equal(MonthStart(today)) {
			spentThisMonth[tx.CategoryId] += tx.Amount
		}
	}
	budgetOf := map[int64]int64{}
	for _, b := range budgets {
		budgetOf[b.CategoryId] = b.Amount
	}

	current := MonthStart(month).Equal(MonthStart(today))
	totals := map[key]int64{}
	for cat, txs := range perCat {
		txs = withoutOneOffs(txs)
		var total int64
		share := map[key]int64{}
		for _, tx := range txs {
			total += tx.Amount
			share[key{tx.AccountId, tx.Business}] += tx.Amount
		}
		if total == 0 {
			continue
		}
		monthly := total / 3
		if b, ok := budgetOf[cat]; ok {
			monthly = b
		}
		if current {
			monthly -= spentThisMonth[cat]
		}
		if monthly <= 0 {
			continue
		}
		for k, s := range share {
			totals[k] += monthly * s / total
		}
	}
	for cat, b := range budgetOf { // budgets for categories with no history
		if _, ok := perCat[cat]; !ok && b > 0 {
			rest := b
			if current {
				rest -= spentThisMonth[cat]
			}
			if rest > 0 {
				totals[key{0, false}] += rest // ponytail: no account history → account 0, shown as "unassigned"
			}
		}
	}

	date := MonthStart(month)
	if current {
		date = today
	}
	var items []Item
	for k, amt := range totals {
		if amt > 0 {
			items = append(items, Item{Date: date, Kind: KindExpense, AccountId: k.acct, Amount: amt, Source: SourceEveryday, Business: k.business, Label: "Everyday spending"})
		}
	}
	sort.Slice(items, func(i, j int) bool { return items[i].AccountId < items[j].AccountId })
	return items
}

func coveredByTemplate(tx Txn, templates []Template) bool {
	for _, t := range templates {
		if t.Kind == tx.Kind && t.AccountId == tx.AccountId && t.CategoryId == tx.CategoryId && within(tx.Amount, t.Amount) {
			return true
		}
	}
	return false
}

func withoutOneOffs(txs []Txn) []Txn {
	amounts := make([]int64, len(txs))
	for i, tx := range txs {
		amounts[i] = tx.Amount
	}
	sort.Slice(amounts, func(i, j int) bool { return amounts[i] < amounts[j] })
	median := amounts[len(amounts)/2]
	var kept []Txn
	for _, tx := range txs {
		if len(txs) < 3 || tx.Amount < oneOffFactor*median {
			kept = append(kept, tx)
		}
	}
	return kept
}
```

Rules: for each statement date S in (today − 1 month, to): billed = (Outstanding for the first S on/after today, else 0) + sum of expense items on the card dated in (previous S, S]. 1. `nextS` = first statement date ≥ today. Due date for statement S = the first date with day `DueDay` strictly after S.
2. Bill 1 at nextS: `Outstanding + Σ card expense items dated (today, nextS]`, paid on due(nextS).
3. Each later statement S_k: `Σ card expense items dated (S_{k-1}, S_k]`, paid on due(S_k).
4. Only bills whose due date < `to` are emitted.
5. A bill is dropped when `repayments` has a transfer into the card dated in (S, due] with amount ≥ 90% of the bill.
6. Emitted item: `Kind: KindTransfer, AccountId: PayFromAccountId, DestinationAccountId: card, Source: SourceCardBill, SourceId: card, Label: "<card label> bill"`.

```go
// pkg/forecast/cards.go
package forecast

import "time"

// Card is a credit card with its billing cycle
type Card struct {
	AccountId, PayFromAccountId int64
	StatementDay, DueDay        int
	Outstanding                 int64
	Label                       string
}

const billPaidShare = 0.9

// CardBills turns card charges into repayment transfers on each due date before `to`
func CardBills(cards []Card, items []Item, repayments []Txn, today, to time.Time) (bills []Item, skipped []int64) {
	for _, c := range cards {
		if c.DueDay < 1 || c.StatementDay < 1 || c.PayFromAccountId == 0 {
			skipped = append(skipped, c.AccountId)
			continue
		}
		prev := today
		carry := c.Outstanding
		for s := statementOnOrAfter(today, c.StatementDay); ; s = statementOnOrAfter(s.AddDate(0, 0, 1), c.StatementDay) {
			due := dayAfter(s, c.DueDay)
			if !due.Before(to) {
				break
			}
			amount := carry
			carry = 0
			for _, it := range items {
				if it.Kind == KindExpense && it.AccountId == c.AccountId && it.Date.After(prev) && !it.Date.After(s) {
					amount += it.Amount
				}
			}
			prev = s
			if amount <= 0 || repaid(repayments, c.AccountId, s, due, amount) {
				continue
			}
			bills = append(bills, Item{Date: due, Kind: KindTransfer, AccountId: c.PayFromAccountId, DestinationAccountId: c.AccountId,
				Amount: amount, Source: SourceCardBill, SourceId: c.AccountId, Label: c.Label + " bill"})
		}
	}
	return bills, skipped
}

func statementOnOrAfter(t time.Time, day int) time.Time {
	s := D(t.Year(), t.Month(), day)
	if s.Before(t) {
		s = s.AddDate(0, 1, 0)
	}
	return s
}

func dayAfter(s time.Time, day int) time.Time {
	d := D(s.Year(), s.Month(), day)
	if !d.After(s) {
		d = d.AddDate(0, 1, 0)
	}
	return d
}

func repaid(repayments []Txn, card int64, from, to time.Time, amount int64) bool {
	for _, r := range repayments {
		if r.DestinationAccountId == card && r.Date.After(from) && !r.Date.After(to) && float64(r.Amount) >= billPaidShare*float64(amount) {
			return true
		}
	}
	return false
}
```

```go
// pkg/forecast/walk.go
package forecast

import (
	"sort"
	"time"
)

type Account struct {
	Id              int64
	Balance         int64
	Cash, Liability bool
	Name            string
}

type Month struct {
	Month                                                 time.Time
	OpeningCash, ClosingCash, ClosingCashWithoutUncertain int64
	BusinessIncome, BusinessCost, PersonalCost            int64
}

type Shortfall struct {
	AccountId     int64
	Date          time.Time
	LowestBalance int64
	LowestDate    time.Time
}

// Walk applies items in date order and summarises each month starting at from's month
func Walk(accounts []Account, items []Item, from time.Time, months int) ([]Month, []Shortfall) {
	sorted := append([]Item(nil), items...)
	sort.SliceStable(sorted, func(i, j int) bool { return sorted[i].Date.Before(sorted[j].Date) })

	byId := map[int64]*Account{}
	bal, balSure := map[int64]int64{}, map[int64]int64{}
	for i := range accounts {
		a := &accounts[i]
		byId[a.Id] = a
		bal[a.Id], balSure[a.Id] = a.Balance, a.Balance
	}
	cash := func(m map[int64]int64) (total int64) {
		for _, a := range accounts {
			if a.Cash {
				total += m[a.Id]
			}
		}
		return
	}
	apply := func(m map[int64]int64, id int64, delta int64) {
		if a, ok := byId[id]; ok && a.Liability {
			m[id] -= delta
		} else {
			m[id] += delta
		}
	}

	sf := map[int64]*Shortfall{}
	var out []Month
	idx := 0
	for k := 0; k < months; k++ {
		start := MonthStart(from).AddDate(0, k, 0)
		end := start.AddDate(0, 1, 0)
		m := Month{Month: start, OpeningCash: cash(bal)}
		for ; idx < len(sorted) && sorted[idx].Date.Before(end); idx++ {
			it := sorted[idx]
			for _, mm := range []map[int64]int64{bal, balSure} {
				if mm == balSure && it.Uncertain {
					continue
				}
				switch it.Kind {
				case KindExpense:
					apply(mm, it.AccountId, -it.Amount)
				case KindIncome:
					apply(mm, it.AccountId, it.Amount)
				case KindTransfer:
					apply(mm, it.AccountId, -it.Amount)
					apply(mm, it.DestinationAccountId, it.Amount)
				}
			}
			switch {
			case it.Kind == KindIncome && it.Business:
				m.BusinessIncome += it.Amount
			case it.Kind == KindExpense && it.Business:
				m.BusinessCost += it.Amount
			case it.Kind == KindExpense:
				m.PersonalCost += it.Amount
			}
			for _, id := range []int64{it.AccountId, it.DestinationAccountId} {
				a, ok := byId[id]
				if !ok || !a.Cash || a.Balance < 0 || balSure[id] >= 0 {
					continue
				}
				if s := sf[id]; s == nil {
					sf[id] = &Shortfall{AccountId: id, Date: it.Date, LowestBalance: balSure[id], LowestDate: it.Date}
				} else if balSure[id] < s.LowestBalance {
					s.LowestBalance, s.LowestDate = balSure[id], it.Date
				}
			}
		}
		m.ClosingCash, m.ClosingCashWithoutUncertain = cash(bal), cash(balSure)
		out = append(out, m)
	}

	var shortfalls []Shortfall
	for _, s := range sf {
		shortfalls = append(shortfalls, *s)
	}
	sort.Slice(shortfalls, func(i, j int) bool { return shortfalls[i].Date.Before(shortfalls[j].Date) })
	return out, shortfalls
}
```

```go
// pkg/forecast/build.go
package forecast

import (
	"sort"
	"time"
)

type ExpectedIncome struct {
	Date      time.Time
	AccountId int64
	Amount    int64
	Source    Source
	SourceId  int64
	Label     string
}

type Input struct {
	Today     time.Time
	Months    int
	Accounts  []Account
	Templates []Template
	Income    []ExpectedIncome
	History   []Txn
	Budgets   []Budget
	Cards     []Card
}

type Result struct {
	Months       []Month
	Items        []Item
	Shortfalls   []Shortfall
	SkippedCards []int64
}

// Build runs the whole forecast
func Build(in Input) Result {
	start := MonthStart(in.Today)
	end := start.AddDate(0, in.Months, 0)

	var planned []Item
	for _, t := range in.Templates {
		planned = append(planned, ExpandTemplate(t, start, end)...)
	}
	planned = DropFulfilled(planned, in.History)
	for i := range planned {
		if planned[i].Date.Before(in.Today) {
			planned[i].Date = in.Today
		}
	}

	for _, inc := range in.Income {
		d := inc.Date
		if d.Before(in.Today) {
			d = in.Today
		}
		if d.Before(end) {
			planned = append(planned, Item{Date: d, Kind: KindIncome, AccountId: inc.AccountId, Amount: inc.Amount,
				Source: inc.Source, SourceId: inc.SourceId, Uncertain: true, Business: true, Label: inc.Label})
		}
	}

	var expenses []Txn
	var repayments []Txn
	for _, tx := range in.History {
		if tx.Kind == KindExpense {
			expenses = append(expenses, tx)
		} else if tx.Kind == KindTransfer && tx.CardRepayment {
			repayments = append(repayments, tx)
		}
	}
	for m := start; m.Before(end); m = m.AddDate(0, 1, 0) {
		planned = append(planned, Everyday(expenses, in.Templates, in.Budgets, in.Today, m)...)
	}

	bills, skipped := CardBills(in.Cards, planned, repayments, in.Today, end)
	planned = append(planned, bills...)
	sort.SliceStable(planned, func(i, j int) bool { return planned[i].Date.Before(planned[j].Date) })

	months, shortfalls := Walk(in.Accounts, planned, in.Today, in.Months)
	return Result{Months: months, Items: planned, Shortfalls: shortfalls, SkippedCards: skipped}
}
```
